Sure! Here's a glassmorphism card using BackdropFilter and ImageFilter.blur — works great with a gradient background.

dart ⧉
class GlassCard extends StatelessWidget { @override Widget build(BuildContext context) { return ClipRRect( borderRadius: BorderRadius.circular(20), child: BackdropFilter( filter: ImageFilter.blur( sigmaX: 12, sigmaY: 12), child: Container( decoration: BoxDecoration( color: Colors.white.withOpacity(0.08), borderRadius: BorderRadius.circular(20), border: Border.all( color: Colors.white.withOpacity(0.15), ), ), padding: EdgeInsets.all(20), ), ), ); } }
